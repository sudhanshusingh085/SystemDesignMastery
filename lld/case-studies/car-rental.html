Design a Car Rental System

Scenario: Design a system like Enterprise or Hertz where users can browse vehicles by location, reserve them for specific dates, and process billing based on different pricing models.

1. What is a Car Rental System?

A physical inventory management system dealing with vehicles moving between different locations (pick-up and drop-off), complex pricing variables (insurance, duration, late fees), and reservation states.

📖 Inventory Management: The process of tracking the availability, location, and condition of items (vehicles, in this case) across multiple branches.

2. What problem does it solve?

It solves the complexity of tracking thousands of vehicles, calculating dynamic pricing rules, and managing the lifecycle of a reservation from pending to active to completed.

4. Java 21 Code Implementation

Java 

import java.util.*;
import java.time.LocalDate;

enum VehicleType { SEDAN, SUV, LUXURY, VAN, TRUCK }
enum ReservationStatus { PENDING, ACTIVE, COMPLETED, CANCELLED }

// OOP: Sealed for fixed hierarchy, pattern matching can be used exhaustively
sealed interface Vehicle permits Sedan, SUV, LuxuryCar, Van, Truck {
    String getLicensePlate();
    double getDailyRate();
}

record Sedan(String licensePlate, double dailyRate) implements Vehicle {
    public String getLicensePlate() { return licensePlate; }
    public double getDailyRate() { return dailyRate; }
}
record SUV(String licensePlate, double dailyRate) implements Vehicle {
    public String getLicensePlate() { return licensePlate; }
    public double getDailyRate() { return dailyRate; }
}
record LuxuryCar(String licensePlate, double dailyRate) implements Vehicle {
    public String getLicensePlate() { return licensePlate; }
    public double getDailyRate() { return dailyRate; }
}
record Van(String licensePlate, double dailyRate) implements Vehicle {
    public String getLicensePlate() { return licensePlate; }
    public double getDailyRate() { return dailyRate; }
}
record Truck(String licensePlate, double dailyRate) implements Vehicle {
    public String getLicensePlate() { return licensePlate; }
    public double getDailyRate() { return dailyRate; }
}

record User(String name, String licenseNumber) {}
record Location(String id, String name, String address) {}

// SOLID: Single Responsibility — this class only handles location-based inventory
class VehicleInventory {
    private Map<Location, List<Vehicle>> inventory = new HashMap<>();
    
    public void addVehicle(Location loc, Vehicle v) {
        inventory.computeIfAbsent(loc, k -> new ArrayList<>()).add(v);
    }
    public List<Vehicle> getAvailable(Location loc) {
        return inventory.getOrDefault(loc, List.of());
    }
}

class Reservation {
    String id; User customer; Vehicle vehicle; Location pickup, returnLoc; 
    LocalDate startDate, endDate; ReservationStatus status;
    
    public Reservation(User customer, Vehicle vehicle, Location pickup, Location returnLoc, LocalDate start, LocalDate end) {
        this.id = UUID.randomUUID().toString(); this.customer = customer; this.vehicle = vehicle;
        this.pickup = pickup; this.returnLoc = returnLoc; this.startDate = start; this.endDate = end;
        this.status = ReservationStatus.PENDING;
    }
}

// Design Pattern: Strategy — interchangeable algorithms for pricing
interface PricingStrategy {
    double calculateTotal(Vehicle vehicle, int days);
}

class StandardPricing implements PricingStrategy {
    public double calculateTotal(Vehicle v, int days) { return v.getDailyRate() * days; }
}

// SOLID: Single Responsibility — calculates bills and late fees
class BillingService {
    public double calculateFinalBill(Reservation r, PricingStrategy strategy, boolean insurance, boolean late) {
        int days = (int) java.time.temporal.ChronoUnit.DAYS.between(r.startDate, r.endDate);
        double total = strategy.calculateTotal(r.vehicle, days);
        if (insurance) total += 20 * days;
        if (late) total += 50; // Late fee flat rate
        return total;
    }
}

// Design Pattern: Observer — notify subscribers on state change
interface RentalObserver {
    void onReservationCreated(Reservation r);
}

class InvoiceGenerator implements RentalObserver {
    public void onReservationCreated(Reservation r) { System.out.println("Invoice generated for " + r.id); }
}

// Design Pattern: Factory
class VehicleFactory {
    public static Vehicle createVehicle(VehicleType type, String plate, double rate) {
        return switch(type) {
            case SEDAN -> new Sedan(plate, rate);
            case SUV -> new SUV(plate, rate);
            case LUXURY -> new LuxuryCar(plate, rate);
            case VAN -> new Van(plate, rate);
            case TRUCK -> new Truck(plate, rate);
        };
    }
}

// Design Pattern: Facade
class RentalService {
    private VehicleInventory inventory = new VehicleInventory();
    private BillingService billing = new BillingService();
    private List<RentalObserver> observers = new ArrayList<>();
    
    public void addObserver(RentalObserver obs) { observers.add(obs); }
    
    public Reservation bookCar(User user, Vehicle v, Location pickup, Location drop, LocalDate start, LocalDate end) {
        Reservation r = new Reservation(user, v, pickup, drop, start, end);
        observers.forEach(o -> o.onReservationCreated(r));
        return r;
    }
}

public class Main {
    public static void main(String[] args) {
        RentalService service = new RentalService();
        service.addObserver(new InvoiceGenerator());
        
        Vehicle v = VehicleFactory.createVehicle(VehicleType.SUV, "XYZ-1234", 100.0);
        Location l1 = new Location("L1", "JFK Airport", "NYC");
        
        service.bookCar(new User("John Doe", "DL12345"), v, l1, l1, LocalDate.now(), LocalDate.now().plusDays(3));
    }
}

5. Why is this recommended?

The Factory Pattern centralizes object creation, making it easy to add new vehicle types. Using Sealed Interfaces ensures that any switch statement checking vehicle types is exhaustive at compile time, preventing runtime bugs.

Interview Tip: Discuss how you would handle inventory consistency across locations. What happens if a car reserved at Location A is returned late to Location B, but another customer has already booked it from A? Mentioning buffer times and fleet repositioning shows deep domain thinking.
