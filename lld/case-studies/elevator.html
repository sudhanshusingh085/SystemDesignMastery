Design an Elevator System

What is an Elevator System?

An elevator system manages the transportation of people between floors in a building. Think of it like a bus network. Buses run on routes, pick up passengers at stops, and drop them off. The system needs to decide which bus (elevator) to send to which stop (floor) to minimize waiting time.

Requirements

  • Building with N floors, M elevators
  • Users press Up/Down button on a floor
  • Inside elevator: press destination floor
  • Elevator scheduling algorithm
  • Display current floor and direction
  • Handle multiple requests efficiently

import java.util.ArrayList;
import java.util.List;
import java.util.PriorityQueue;

enum Direction { UP, DOWN, IDLE }

record ElevatorRequest(int fromFloor, Direction direction) {}
record InternalRequest(int toFloor) {}

// Design Pattern: Strategy
interface ElevatorSchedulingStrategy {
    Elevator selectElevator(List<Elevator> elevators, ElevatorRequest request);
}

class NearestElevatorStrategy implements ElevatorSchedulingStrategy {
    public Elevator selectElevator(List<Elevator> elevators, ElevatorRequest request) {
        Elevator best = elevators.get(0);
        int minDistance = Integer.MAX_VALUE;
        for (Elevator e : elevators) {
            int dist = Math.abs(e.getCurrentFloor() - request.fromFloor());
            if (dist < minDistance) {
                minDistance = dist;
                best = e;
            }
        }
        return best;
    }
}

class ZonedStrategy implements ElevatorSchedulingStrategy {
    public Elevator selectElevator(List<Elevator> elevators, ElevatorRequest request) {
        // Implementation for zoned strategy
        return elevators.get(0);
    }
}

class Elevator {
    private final int id;
    private int currentFloor;
    private Direction direction;
    private PriorityQueue<Integer> upQueue;
    private PriorityQueue<Integer> downQueue;

    public Elevator(int id) {
        this.id = id;
        this.currentFloor = 0;
        this.direction = Direction.IDLE;
        this.upQueue = new PriorityQueue<>();
        this.downQueue = new PriorityQueue<>((a, b) -> b.compareTo(a));
    }

    public int getCurrentFloor() { return currentFloor; }
    public Direction getDirection() { return direction; }

    public void addRequest(InternalRequest req) {
        if (req.toFloor() > currentFloor) {
            upQueue.offer(req.toFloor());
        } else {
            downQueue.offer(req.toFloor());
        }
    }

    public void addExternalRequest(ElevatorRequest req) {
        if (req.fromFloor() > currentFloor) {
            upQueue.offer(req.fromFloor());
        } else {
            downQueue.offer(req.fromFloor());
        }
    }

    public void move() {
        if (!upQueue.isEmpty()) {
            direction = Direction.UP;
            currentFloor = upQueue.poll();
            System.out.println("Elevator " + id + " moved UP to floor " + currentFloor);
        } else if (!downQueue.isEmpty()) {
            direction = Direction.DOWN;
            currentFloor = downQueue.poll();
            System.out.println("Elevator " + id + " moved DOWN to floor " + currentFloor);
        } else {
            direction = Direction.IDLE;
            System.out.println("Elevator " + id + " is IDLE at floor " + currentFloor);
        }
    }
}

// SOLID: DIP
class ElevatorController {
    private final List<Elevator> elevators;
    private final ElevatorSchedulingStrategy strategy;

    public ElevatorController(List<Elevator> elevators, ElevatorSchedulingStrategy strategy) {
        this.elevators = elevators;
        this.strategy = strategy;
    }

    public void requestElevator(ElevatorRequest request) {
        Elevator selected = strategy.selectElevator(elevators, request);
        selected.addExternalRequest(request);
        System.out.println("Assigned Elevator to request at floor " + request.fromFloor());
    }
}

// Design Pattern: Facade
class Building {
    private final List<Elevator> elevators;
    private final ElevatorController controller;

    public Building(int numElevators) {
        elevators = new ArrayList<>();
        for (int i = 0; i < numElevators; i++) {
            elevators.add(new Elevator(i));
        }
        controller = new ElevatorController(elevators, new NearestElevatorStrategy());
    }

    public void callElevator(int floor, Direction direction) {
        controller.requestElevator(new ElevatorRequest(floor, direction));
    }

    public void step() {
        for (Elevator e : elevators) {
            e.move();
        }
    }
}

// SOLID: ISP — different interfaces for different panels
interface Panel { }
class FloorPanel implements Panel { }
class ElevatorPanel implements Panel { }
interface Display { void show(); }

public class Main {
    public static void main(String[] args) {
        Building building = new Building(2);
        building.callElevator(5, Direction.UP);
        building.step();
        building.step();
    }
}
