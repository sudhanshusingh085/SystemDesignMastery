Design Tic-Tac-Toe (LLD)

Scenario: Design a flexible Tic-Tac-Toe game supporting an NxN board, turn-based play, and win/draw detection.

1. What is Tic-Tac-Toe System?

A simpler board game, but designing it flexibly means it should easily scale from a 3x3 grid to an NxN grid. We need a way to track the

📖 Grid: A 2D array holding the state of the board.
and dynamically evaluate win conditions.

2. What problem does it solve?

  • Scaling: Hardcoding checks for exactly 3 rows/cols is brittle.
  • Modularity: Decoupling input handling from game logic.

4. Java 21 Implementation

import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

enum Symbol { X, O, EMPTY }

// Record for immutable grid coordinates
record Cell(int row, int col) {}

// Record for historical moves (used for Undo functionality)
record Move(Cell cell, Symbol symbol) {}

// SOLID: SRP — The board only stores state and knows if it's full.
class Board {
    private final int size;
    private final Symbol[][] grid;

    public Board(int size) {
        this.size = size;
        this.grid = new Symbol[size][size];
        for (int i = 0; i < size; i++) {
            for (int j = 0; j < size; j++) {
                grid[i][j] = Symbol.EMPTY;
            }
        }
    }

    public int getSize() { return size; }
    public Symbol getSymbol(int r, int c) { return grid[r][c]; }

    public boolean placeSymbol(Cell cell, Symbol symbol) {
        if (cell.row() < 0 || cell.row() >= size || cell.col() < 0 || cell.col() >= size) return false;
        if (grid[cell.row()][cell.col()] != Symbol.EMPTY) return false;
        
        grid[cell.row()][cell.col()] = symbol;
        return true;
    }

    public boolean isFull() {
        for (int i = 0; i < size; i++) {
            for (int j = 0; j < size; j++) {
                if (grid[i][j] == Symbol.EMPTY) return false;
            }
        }
        return true;
    }

    public void display() {
        for (int i = 0; i < size; i++) {
            for (int j = 0; j < size; j++) {
                System.out.print((grid[i][j] == Symbol.EMPTY ? "-" : grid[i][j]) + " ");
            }
            System.out.println();
        }
    }
}

// SOLID: OCP — New win rules (like a 'Corners Win' variant) can be added without modifying the core Game.
interface WinChecker {
    boolean checkWin(Board board, Symbol symbol);
}

class StandardWinChecker implements WinChecker {
    @Override
    public boolean checkWin(Board board, Symbol symbol) {
        int n = board.getSize();
        
        // Check rows and cols
        for (int i = 0; i < n; i++) {
            boolean rowWin = true, colWin = true;
            for (int j = 0; j < n; j++) {
                if (board.getSymbol(i, j) != symbol) rowWin = false;
                if (board.getSymbol(j, i) != symbol) colWin = false;
            }
            if (rowWin || colWin) return true;
        }

        // Check diagonals
        boolean diag1 = true, diag2 = true;
        for (int i = 0; i < n; i++) {
            if (board.getSymbol(i, i) != symbol) diag1 = false;
            if (board.getSymbol(i, n - 1 - i) != symbol) diag2 = false;
        }

        return diag1 || diag2;
    }
}

class Player {
    private String name;
    private Symbol symbol;
    
    public Player(String name, Symbol symbol) {
        this.name = name;
        this.symbol = symbol;
    }
    public Symbol getSymbol() { return symbol; }
    public String getName() { return name; }
}

// Sealed interfaces guarantee exhaustive pattern matching checks in Java 21+ switch statements
sealed interface GameState permits Playing, Won, Draw {}
record Playing(Player currentPlayer) implements GameState {}
record Won(Player winner) implements GameState {}
record Draw() implements GameState {}

// SOLID: DIP — We depend on an abstraction for input, not concrete console Scanners.
interface InputHandler {
    Cell getMove(Player p);
}

class ConsoleInputHandler implements InputHandler {
    private Scanner scanner = new Scanner(System.in);
    @Override
    public Cell getMove(Player p) {
        System.out.println(p.getName() + "'s turn. Enter row and col:");
        int r = scanner.nextInt();
        int c = scanner.nextInt();
        return new Cell(r, c);
    }
}

class GameHistory {
    private List moves = new ArrayList<>();
    public void addMove(Move m) { moves.add(m); }

// Design Pattern: Facade & Template Method for the game loop
class TicTacToeGame {
    private Board board;
    private Player[] players;
    private int turn;
    private WinChecker winChecker;
    private InputHandler inputHandler;
    private GameState state;

    public TicTacToeGame(int size, Player p1, Player p2, InputHandler inputHandler) {
        this.board = new Board(size);
        this.players = new Player[]{p1, p2};
        this.turn = 0;
        this.winChecker = new StandardWinChecker();
        this.inputHandler = inputHandler;
        this.state = new Playing(players[0]);
    }

    public void start() {
        while (state instanceof Playing playingState) {
            board.display();
            Player current = playingState.currentPlayer();
            Cell cell = inputHandler.getMove(current);
            
            if (!board.placeSymbol(cell, current.getSymbol())) {
                System.out.println("Invalid move, try again.");
                continue;
            }

            if (winChecker.checkWin(board, current.getSymbol())) {
                state = new Won(current);
            } else if (board.isFull()) {
                state = new Draw();
            } else {
                turn = 1 - turn;
                state = new Playing(players[turn]);
            }
        }
        
        board.display();
        
        // Java 21 Pattern Matching for switch
        switch (state) {
            case Won w -> System.out.println(w.winner().getName() + " wins!");
            case Draw d -> System.out.println("It's a draw!");
            case Playing p -> {} // unreachable
        }
    }
}

using System;
using System.Collections.Generic;

public enum Symbol { Empty, X, O }

public class Board
{
    private readonly Symbol[,] _grid;
    public int Size { get; }

    public Board(int size = 3)
    {
        Size = size;
        _grid = new Symbol[size, size];
    }

    public bool PlaceSymbol(int row, int col, Symbol symbol)
    {
        if (row < 0 || row >= Size || col < 0 || col >= Size || _grid[row, col] != Symbol.Empty)
        {
            return false;
        }
        _grid[row, col] = symbol;
        return true;
    }

    public Symbol GetSymbol(int row, int col) => _grid[row, col];

    public bool IsFull()
    {
        for (int r = 0; r < Size; r++)
            for (int c = 0; c < Size; c++)
                if (_grid[r, c] == Symbol.Empty) return false;
        return true;
    }

    public void PrintBoard()
    {
        for (int r = 0; r < Size; r++)
        {
            for (int c = 0; c < Size; c++)
            {
                char ch = _grid[r, c] switch
                {
                    Symbol.X => 'X',
                    Symbol.O => 'O',
                    _ => '.'
                };
                Console.Write($" {ch} ");
                if (c < Size - 1) Console.Write("|");
            }
            Console.WriteLine();
            if (r < Size - 1) Console.WriteLine("---+---+---");
        }
    }
}

public interface IWinChecker
{
    bool CheckWin(Board board, int lastRow, int lastCol, Symbol symbol);
}

public class StandardWinChecker : IWinChecker
{
    public bool CheckWin(Board board, int row, int col, Symbol symbol)
    {
        int size = board.Size;

        // Check row
        bool win = true;
        for (int c = 0; c < size; c++)
        {
            if (board.GetSymbol(row, c) != symbol) { win = false; break; }
        }
        if (win) return true;

        // Check col
        win = true;
        for (int r = 0; r < size; r++)
        {
            if (board.GetSymbol(r, col) != symbol) { win = false; break; }
        }
        if (win) return true;

        // Check main diagonal
        if (row == col)
        {
            win = true;
            for (int i = 0; i < size; i++)
            {
                if (board.GetSymbol(i, i) != symbol) { win = false; break; }
            }
            if (win) return true;
        }

        // Check anti-diagonal
        if (row + col == size - 1)
        {
            win = true;
            for (int i = 0; i < size; i++)
            {
                if (board.GetSymbol(i, size - 1 - i) != symbol) { win = false; break; }
            }
            if (win) return true;
        }

        return false;
    }
}

public record Player(string Name, Symbol Symbol);

public class TicTacToeGame
{
    private readonly Board _board;
    private readonly Queue _players = new();
    private readonly IWinChecker _winChecker;

    public TicTacToeGame(Player p1, Player p2, int size = 3)
    {
        _board = new Board(size);
        _players.Enqueue(p1);
        _players.Enqueue(p2);
        _winChecker = new StandardWinChecker();
    }

    public void Play()
    {
        bool gameOver = false;
        while (!gameOver)
        {
            _board.PrintBoard();
            var current = _players.Dequeue();
            Console.WriteLine($"[Turn] {current.Name} ({current.Symbol}) - enter move (simulated):");

            // Simple simulated turn for demonstration
            int row = 0, col = 0;
            for (int r = 0; r < _board.Size; r++)
            {
                for (int c = 0; c < _board.Size; c++)
                {
                    if (_board.GetSymbol(r, c) == Symbol.Empty)
                    {
                        row = r; col = c; goto MoveFound;
                    }
                }
            }
        MoveFound:
            _board.PlaceSymbol(row, col, current.Symbol);
            Console.WriteLine($"{current.Name} placed at ({row}, {col})");

            if (_winChecker.CheckWin(_board, row, col, current.Symbol))
            {
                _board.PrintBoard();
                Console.WriteLine($"🎉 GAME OVER! Winner: {current.Name}!");
                gameOver = true;
            }
            else if (_board.IsFull())
            {
                _board.PrintBoard();
                Console.WriteLine("🤝 GAME OVER! It's a DRAW!");
                gameOver = true;
            }
            else
            {
                _players.Enqueue(current);
            }
        }
    }
}

public class Program
{
    public static void Main()
    {
        var game = new TicTacToeGame(new Player("Alice", Symbol.X), new Player("Bob", Symbol.O));
        game.Play();
    }
}

5. Why is it recommended / When to use it?

By extracting WinChecker, we handle an NxN board without deeply nested, hardcoded arrays. The InputHandler injection ensures we can easily adapt the game for a web UI or mobile app without changing the core game logic.

Interview Tip: Interviewers love scalable implementations. Always ask: "Does the board have to be 3x3?" Designing it for NxN from the start scores major points.
