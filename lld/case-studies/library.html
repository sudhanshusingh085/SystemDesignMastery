Design a Library System

What is a Library System?

A library system manages books, members, and the borrowing/returning process. Think of it like a rental service. People can browse inventory, take items for a limited time, and return them, paying a fine if they are late.

Requirements

  • Add/remove books
  • Search books by title, author, ISBN
  • Member registration
  • Borrow/return books
  • Fine calculation for late returns
  • Maximum 5 books per member

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

// Immutable data
record BookInfo(String isbn, String title, String author, int year) {}

class Book {
    private final BookInfo info;
    private boolean available;

    public Book(BookInfo info) {
        this.info = info;
        this.available = true;
    }

    public BookInfo getInfo() { return info; }
    public boolean isAvailable() { return available; }
    public void setAvailable(boolean available) { this.available = available; }
}

sealed interface MemberType permits RegularMember, PremiumMember {}
record RegularMember() implements MemberType {}
record PremiumMember() implements MemberType {}

class Member {
    private final String name;
    private final String memberId;
    private final List<Book> borrowedBooks;
    private final MemberType memberType;
    private static final int MAX_BOOKS = 5;

    public Member(String name, String memberId, MemberType type) {
        this.name = name;
        this.memberId = memberId;
        this.memberType = type;
        this.borrowedBooks = new ArrayList<>();
    }

    public boolean canBorrow() { return borrowedBooks.size() < MAX_BOOKS; }
    public void borrowBook(Book book) { borrowedBooks.add(book); }
    public void returnBook(Book book) { borrowedBooks.remove(book); }
    public String getMemberId() { return memberId; }
}

// Design Pattern: Strategy
interface SearchStrategy {
    List<Book> search(List<Book> books, String query);
}

class TitleSearch implements SearchStrategy {
    public List<Book> search(List<Book> books, String query) {
        return books.stream().filter(b -> b.getInfo().title().contains(query)).collect(Collectors.toList());
    }
}

class AuthorSearch implements SearchStrategy {
    public List<Book> search(List<Book> books, String query) {
        return books.stream().filter(b -> b.getInfo().author().contains(query)).collect(Collectors.toList());
    }
}

class IsbnSearch implements SearchStrategy {
    public List<Book> search(List<Book> books, String query) {
        return books.stream().filter(b -> b.getInfo().isbn().equals(query)).collect(Collectors.toList());
    }
}

// SOLID: DIP
interface FineCalculator {
    double calculateFine(long overdueDays);
}

class DailyFineCalculator implements FineCalculator {
    public double calculateFine(long overdueDays) {
        return overdueDays > 0 ? overdueDays * 1.5 : 0;
    }
}

record BorrowRecord(String memberId, String isbn, LocalDate borrowDate, LocalDate dueDate, LocalDate returnDate) {}

// Design Pattern: Observer
interface LibraryObserver {
    void onBookBorrowed(String memberId, String isbn);
    void onBookReturned(String memberId, String isbn);
}

class NotificationService implements LibraryObserver {
    public void onBookBorrowed(String memberId, String isbn) {
        System.out.println("Notification: Member " + memberId + " borrowed book " + isbn);
    }
    public void onBookReturned(String memberId, String isbn) {
        System.out.println("Notification: Member " + memberId + " returned book " + isbn);
    }
}

// Design Pattern: Facade
class Library {
    private final List<Book> catalog = new ArrayList<>();
    private final List<Member> members = new ArrayList<>();
    private final List<LibraryObserver> observers = new ArrayList<>();

    public void addBook(Book book) { catalog.add(book); }
    public void registerMember(Member member) { members.add(member); }
    
    public void addObserver(LibraryObserver observer) { observers.add(observer); }

    public void borrowBook(String memberId, String isbn) {
        Member member = members.stream().filter(m -> m.getMemberId().equals(memberId)).findFirst().orElse(null);
        Book book = catalog.stream().filter(b -> b.getInfo().isbn().equals(isbn) && b.isAvailable()).findFirst().orElse(null);

        if (member != null && book != null && member.canBorrow()) {
            book.setAvailable(false);
            member.borrowBook(book);
            for (LibraryObserver obs : observers) {
                obs.onBookBorrowed(memberId, isbn);
            }
        } else {
            System.out.println("Cannot borrow book.");
        }
    }
}

public class Main {
    public static void main(String[] args) {
        Library lib = new Library();
        lib.addObserver(new NotificationService());

        Book book = new Book(new BookInfo("123", "Design Patterns", "GoF", 1994));
        lib.addBook(book);

        Member member = new Member("Alice", "M01", new RegularMember());
        lib.registerMember(member);

        lib.borrowBook("M01", "123");
    }
}
    
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Linq;

namespace LibrarySystem;

public enum BookStatus { Available, Loaned, Lost }

public record BookItem(string Barcode, string Title, string Author, string Isbn)
{
    public BookStatus Status { get; set; } = BookStatus.Available;
}

public record Member(string MemberId, string Name, string Email);

public class LoanRecord(string loanId, string barcode, string memberId, DateTime issueDate, DateTime dueDate)
{
    public string LoanId { get; } = loanId;
    public string Barcode { get; } = barcode;
    public string MemberId { get; } = memberId;
    public DateTime IssueDate { get; } = issueDate;
    public DateTime DueDate { get; } = dueDate;
    public DateTime? ReturnDate { get; set; }
}

public class LibraryService
{
    private readonly ConcurrentDictionary _books = new();
    private readonly ConcurrentDictionary _members = new();
    private readonly ConcurrentDictionary _activeLoans = new();
    private const int MaxBorrowLimit = 5;
    private const int LoanDurationDays = 14;

    public void AddBook(BookItem book) => _books[book.Barcode] = book;
    public void RegisterMember(Member member) => _members[member.MemberId] = member;

    public LoanRecord? CheckoutBook(string barcode, string memberId)
    {
        if (!_books.TryGetValue(barcode, out var book) || book.Status != BookStatus.Available)
            return null;

        int currentLoanCount = _activeLoans.Values.Count(l => l.MemberId == memberId && l.ReturnDate == null);
        if (currentLoanCount >= MaxBorrowLimit)
            return null;

        lock (book)
        {
            if (book.Status != BookStatus.Available) return null;
            book.Status = BookStatus.Loaned;

            var loan = new LoanRecord(
                Guid.NewGuid().ToString()[..8],
                barcode,
                memberId,
                DateTime.UtcNow,
                DateTime.UtcNow.AddDays(LoanDurationDays)
            );
            _activeLoans[loan.LoanId] = loan;
            return loan;
        }
    }

    public bool ReturnBook(string loanId)
    {
        if (!_activeLoans.TryGetValue(loanId, out var loan) || loan.ReturnDate != null)
            return false;

        if (_books.TryGetValue(loan.Barcode, out var book))
        {
            lock (book)
            {
                loan.ReturnDate = DateTime.UtcNow;
                book.Status = BookStatus.Available;
                return true;
            }
        }
        return false;
    }
}
