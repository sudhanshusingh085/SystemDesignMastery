Design a Ride Sharing System (Uber/Lyft)

What is a Ride Sharing System?

A ride-sharing system matches a rider needing a lift with a nearby driver willing to drive them. It calculates the price based on distance and demand, tracks the trip in real-time, and handles payments.

📖 Surge Pricing: A dynamic pricing model where prices increase when demand for rides is higher than the supply of available drivers in an area.

What Problem Does It Solve?

Matching algorithms must be fast and geographical. Pricing must be flexible and dynamic. Trip states (requested, driver assigned, in progress) must be strictly managed so a trip isn't marked "complete" before the driver arrives.

How Does It Work?

The core uses the Strategy pattern heavily: one strategy for matching drivers (nearest vs highest rated), and another strategy (plus Decorator) for pricing to dynamically wrap base fares with surge multipliers. The State pattern manages the trip lifecycle.

Low-Level Design (Java 21)

import java.time.LocalDateTime;
import java.util.*;

record Location(double latitude, double longitude) {
    public double distanceTo(Location other) {
        // Simplified distance calculation
        return Math.sqrt(Math.pow(latitude - other.latitude, 2) + Math.pow(longitude - other.longitude, 2));
    }
}

record RiderId(String id) {}
record DriverId(String id) {}

class Rider {
    private RiderId id;
    private String name;
    private Location currentLocation;
    // getters/setters...
}

enum VehicleType { BIKE, AUTO, MINI, SEDAN, SUV, PREMIUM }

class Driver {
    private DriverId id;
    private String name;
    private Location currentLocation;
    private VehicleType vehicleType;
    private double rating;
    private boolean isAvailable;
    
    public Driver(DriverId id, Location loc, double rating) {
        this.id = id;
        this.currentLocation = loc;
        this.rating = rating;
        this.isAvailable = true;
    }
    public Location getCurrentLocation() { return currentLocation; }
    public double getRating() { return rating; }
    public void setAvailable(boolean b) { isAvailable = b; }
}

// Design Pattern: State — Strict trip lifecycle
sealed interface TripStatus permits Requested, DriverAssigned, InProgress, Completed, Cancelled {}
record Requested() implements TripStatus {}
record DriverAssigned() implements TripStatus {}
record InProgress() implements TripStatus {}
record Completed() implements TripStatus {}
record Cancelled() implements TripStatus {}

// OOP: Composition — Trip owns TripStatus lifecycle
class Trip {
    private final String tripId;
    private final Rider rider;
    private Driver driver;
    private final Location source;
    private final Location destination;
    private TripStatus status;
    private double fare;
    
    public Trip(String tripId, Rider rider, Location source, Location destination) {
        this.tripId = tripId;
        this.rider = rider;
        this.source = source;
        this.destination = destination;
        this.status = new Requested();
    }
    
    public void setDriver(Driver driver) { this.driver = driver; }
    public void setStatus(TripStatus s) { this.status = s; }
    public void setFare(double fare) { this.fare = fare; }
    
    public Location getSource() { return source; }
    public Location getDestination() { return destination; }
    public TripStatus getStatus() { return status; }
}

// Design Pattern: State
class TripStateMachine {
    public void transition(Trip trip, TripStatus newStatus) {
        boolean valid = switch(trip.getStatus()) {
            case Requested r -> newStatus instanceof DriverAssigned || newStatus instanceof Cancelled;
            case DriverAssigned d -> newStatus instanceof InProgress || newStatus instanceof Cancelled;
            case InProgress i -> newStatus instanceof Completed;
            case Completed c -> false;
            case Cancelled c -> false;
        };
        if(valid) trip.setStatus(newStatus);
        else throw new IllegalStateException("Invalid state transition");
    }
}

// Design Pattern: Strategy
interface DriverMatchingStrategy {
    Driver matchDriver(Trip trip, List<Driver> availableDrivers);
}

class NearestDriverMatcher implements DriverMatchingStrategy {
    @Override
    public Driver matchDriver(Trip trip, List<Driver> availableDrivers) {
        return availableDrivers.stream()
                .min(Comparator.comparingDouble(d -> d.getCurrentLocation().distanceTo(trip.getSource())))
                .orElse(null);
    }
}

class HighestRatedDriverMatcher implements DriverMatchingStrategy {
    @Override
    public Driver matchDriver(Trip trip, List<Driver> availableDrivers) {
        return availableDrivers.stream()
                .max(Comparator.comparingDouble(Driver::getRating))
                .orElse(null);
    }
}

// Design Pattern: Strategy
interface PricingStrategy {
    double calculateFare(Location source, Location destination);
}

class BasePricing implements PricingStrategy {
    @Override
    public double calculateFare(Location source, Location destination) {
        double distance = source.distanceTo(destination);
        return 5.0 + (distance * 2.0); // Base fee + per unit distance
    }
}

// Design Pattern: Decorator wrapping BasePricing
class SurgePricing implements PricingStrategy {
    private final PricingStrategy baseStrategy;
    private final double surgeMultiplier;
    
    public SurgePricing(PricingStrategy baseStrategy, double surgeMultiplier) {
        this.baseStrategy = baseStrategy;
        this.surgeMultiplier = surgeMultiplier;
    }
    
    @Override
    public double calculateFare(Location source, Location destination) {
        return baseStrategy.calculateFare(source, destination) * surgeMultiplier;
    }
}

// SOLID: SRP — manages only driver availability
class DriverPool {
    private final List<Driver> activeDrivers = new ArrayList<>();
    
    public void addDriver(Driver d) { activeDrivers.add(d); }
    public List<Driver> getAvailableDrivers() {
        return activeDrivers.stream().filter(d -> true).toList(); // Simplified
    }
}

// Design Pattern: Observer
interface TripObserver {
    void onTripUpdated(Trip trip);
}
class RiderNotifier implements TripObserver {
    public void onTripUpdated(Trip trip) { System.out.println("Rider UI update: " + trip.getStatus()); }
}

// Design Pattern: Facade
class RideService {
    private final TripStateMachine stateMachine = new TripStateMachine();
    private final DriverMatchingStrategy matchingStrategy;
    private final PricingStrategy pricingStrategy;
    private final DriverPool driverPool;
    private final List<TripObserver> observers = new ArrayList<>();
    
    public RideService(DriverMatchingStrategy ms, PricingStrategy ps, DriverPool pool) {
        this.matchingStrategy = ms;
        this.pricingStrategy = ps;
        this.driverPool = pool;
    }
    
    public void addObserver(TripObserver o) { observers.add(o); }
    
    public Trip requestRide(Rider rider, Location dest) {
        Trip trip = new Trip(UUID.randomUUID().toString(), rider, new Location(0,0), dest);
        trip.setFare(pricingStrategy.calculateFare(new Location(0,0), dest));
        
        Driver driver = matchingStrategy.matchDriver(trip, driverPool.getAvailableDrivers());
        if(driver != null) {
            trip.setDriver(driver);
            driver.setAvailable(false);
            stateMachine.transition(trip, new DriverAssigned());
        }
        notify(trip);
        return trip;
    }
    
    private void notify(Trip t) { observers.forEach(o -> o.onTripUpdated(t)); }
}

Interview Tip

Using the Decorator Pattern for Surge Pricing is a massive flex. It shows you know how to wrap base logic dynamically at runtime without modifying the underlying pricing classes.
