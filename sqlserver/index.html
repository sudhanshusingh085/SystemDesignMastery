CREATE DATABASE FinancialTradingDB;
GO
USE FinancialTradingDB;
GO

-- 1. Clients Table (Institutional / Retail)
CREATE TABLE Clients (
    ClientID INT IDENTITY(1,1) PRIMARY KEY,
    ClientName NVARCHAR(150) NOT NULL,
    ClientType VARCHAR(20) NOT NULL CHECK (ClientType IN ('INSTITUTIONAL', 'HEDGE_FUND', 'RETAIL', 'MARKET_MAKER')),
    CountryCode CHAR(2) NOT NULL DEFAULT 'US',
    KYCStatus VARCHAR(20) NOT NULL DEFAULT 'VERIFIED',
    CreatedAt DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME()
);

-- 2. Trading Accounts
CREATE TABLE Accounts (
    AccountID INT IDENTITY(1001,1) PRIMARY KEY,
    ClientID INT NOT NULL FOREIGN KEY REFERENCES Clients(ClientID),
    AccountNumber VARCHAR(30) NOT NULL UNIQUE,
    AccountType VARCHAR(20) NOT NULL CHECK (AccountType IN ('CASH', 'MARGIN', 'DVP_SETTLEMENT')),
    Currency CHAR(3) NOT NULL DEFAULT 'USD',
    CashBalance DECIMAL(18,4) NOT NULL DEFAULT 0.0000,
    MarginLimit DECIMAL(18,4) NOT NULL DEFAULT 0.0000,
    Status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    UpdatedAt DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME()
);

-- 3. Securities Master
CREATE TABLE Securities (
    SecurityID INT IDENTITY(5001,1) PRIMARY KEY,
    Ticker VARCHAR(10) NOT NULL UNIQUE,
    SecurityName NVARCHAR(100) NOT NULL,
    AssetClass VARCHAR(20) NOT NULL CHECK (AssetClass IN ('EQUITY', 'BOND', 'FX', 'COMMODITY', 'CRYPTO')),
    Sector VARCHAR(50) NULL,
    LotSize INT NOT NULL DEFAULT 1,
    IsActive BIT NOT NULL DEFAULT 1
);

-- 4. Orders
CREATE TABLE Orders (
    OrderID BIGINT IDENTITY(100001,1) PRIMARY KEY,
    AccountID INT NOT NULL FOREIGN KEY REFERENCES Accounts(AccountID),
    SecurityID INT NOT NULL FOREIGN KEY REFERENCES Securities(SecurityID),
    OrderSide VARCHAR(4) NOT NULL CHECK (OrderSide IN ('BUY', 'SELL')),
    OrderType VARCHAR(10) NOT NULL CHECK (OrderType IN ('LIMIT', 'MARKET', 'STOP_LOSS')),
    Quantity INT NOT NULL CHECK (Quantity > 0),
    LimitPrice DECIMAL(18,4) NULL,
    FilledQuantity INT NOT NULL DEFAULT 0,
    OrderStatus VARCHAR(20) NOT NULL DEFAULT 'PENDING' 
        CHECK (OrderStatus IN ('PENDING', 'PARTIAL', 'FILLED', 'CANCELLED', 'REJECTED')),
    OrderTime DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME(),
    VersionRowChange ROWVERSION -- For optimistic locking
);

-- 5. Executions / Fills
CREATE TABLE Executions (
    ExecutionID BIGINT IDENTITY(500001,1) PRIMARY KEY,
    OrderID BIGINT NOT NULL FOREIGN KEY REFERENCES Orders(OrderID),
    ExecutionVenue VARCHAR(20) NOT NULL DEFAULT 'NYSE',
    FillQuantity INT NOT NULL CHECK (FillQuantity > 0),
    FillPrice DECIMAL(18,4) NOT NULL,
    LiquidityFlag CHAR(1) NOT NULL CHECK (LiquidityFlag IN ('A', 'R')), -- Added / Removed
    ExecutionFee DECIMAL(10,4) NOT NULL DEFAULT 0.0000,
    ExecutedAt DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME()
);

-- 6. Current Portfolio Positions
CREATE TABLE PortfolioPositions (
    PositionID INT IDENTITY(1,1) PRIMARY KEY,
    AccountID INT NOT NULL FOREIGN KEY REFERENCES Accounts(AccountID),
    SecurityID INT NOT NULL FOREIGN KEY REFERENCES Securities(SecurityID),
    PositionUnits INT NOT NULL,
    AverageEntryPrice DECIMAL(18,4) NOT NULL,
    CurrentMarketPrice DECIMAL(18,4) NOT NULL,
    LastValuationTime DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT UQ_Account_Security UNIQUE (AccountID, SecurityID)
);

-- 7. Ledger Transactions (Double-entry Cash / Stock Transfer)
CREATE TABLE LedgerTransactions (
    TransactionID BIGINT IDENTITY(800001,1) PRIMARY KEY,
    AccountID INT NOT NULL FOREIGN KEY REFERENCES Accounts(AccountID),
    TransactionType VARCHAR(20) NOT NULL CHECK (TransactionType IN ('DEPOSIT', 'WITHDRAWAL', 'TRADE_BUY', 'TRADE_SELL', 'DIVIDEND', 'FEE')),
    Amount DECIMAL(18,4) NOT NULL,
    TransactionTimestamp DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME(),
    ReferenceNote NVARCHAR(250) NULL
);

-- 8. Audit Trail
CREATE TABLE AuditLogs (
    LogID BIGINT IDENTITY(1,1) PRIMARY KEY,
    TableName VARCHAR(50) NOT NULL,
    OperationType CHAR(6) NOT NULL CHECK (OperationType IN ('INSERT', 'UPDATE', 'DELETE')),
    RecordID VARCHAR(50) NOT NULL,
    ChangedBy NVARCHAR(100) NOT NULL DEFAULT SUSER_SNAME(),
    ChangePayload NVARCHAR(MAX) NULL,
    LoggedAt DATETIME2(3) NOT NULL DEFAULT SYSUTCDATETIME()
);
